cmake_minimum_required(VERSION 3.22.1)
project(rimoat_encoder)
foreach(ABI arm64-v8a armeabi-v7a)
    file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../jniLibs/${ABI}/liblocalirdb.so" LIB_HASH)
    if(ABI STREQUAL "arm64-v8a")
        set(EXPECTED_HASH "1c988e5eb4f7a13ebee336a2678f38dbbfc1715e7cda761abfb877f4f8b1d59d")
    else()
        set(EXPECTED_HASH "61e67cbb2cb34f89af412f9cd07490f7400284a80f9919d248e2d73b7c382d28")
    endif()
    if(NOT LIB_HASH STREQUAL EXPECTED_HASH)
        message(FATAL_ERROR "Local IR encoder binary changed: re-verify offsets for ${ABI}")
    endif()
endforeach()
add_library(rimoat_encoder SHARED ac_encoder.cpp)
target_link_libraries(rimoat_encoder dl)
target_compile_features(rimoat_encoder PRIVATE cxx_std_17)
